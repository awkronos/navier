  {u v f : Navier.Construction.ProblemStatement.VelocityField}
  {p q : Navier.Construction.ProblemStatement.PressureField},
  Navier.Construction.R3CompactCandidate.Properties u p f →
    ∀ {T : ℝ},
      T < 1 →
        ContDiffOn ℝ (↑⊤) v (Navier.ConstructionR3.Comparison.slab 0 T) →
          ContDiffOn ℝ (↑⊤) q (Navier.ConstructionR3.Comparison.slab 0 T) →
            Navier.ConstructionR3.ProblemStatement.UniformFiniteEnergy (Set.Icc 0 T) v →
              (∀ t ∈ Set.Ioo 0 T,
                  ∀ (x : Navier.Construction.ProblemStatement.Space),
                    Navier.Construction.ProblemStatement.spatialDivergence v t x = 0) →
                (∀ t ∈ Set.Ioo 0 T,
                    ∀ (x : Navier.Construction.ProblemStatement.Space),
                      Navier.Construction.ProblemStatement.navierStokesResidual v q t x = f (t, x)) →
                  (∀ (x : Navier.Construction.ProblemStatement.Space), v (0, x) = 0) →
                    ∀ t ∈ Set.Icc 0 T, ∀ (x : Navier.Construction.ProblemStatement.Space), u (t, x) = v (t, x) :=
fun {u v f} {p q} h {T} hT1 hv hq hev hdv hNSv hvzero =>
  if hTpos : 0 < T then
    have hpre := fun ⦃z⦄ hz => ⟨⟨hz.left.left, LE.le.trans_lt hz.left.right hT1⟩, hz.right⟩;
    Exists.casesOn h.velocity_support fun K h_1 =>
      And.casesOn h_1 fun hK hs =>
        have hsupport := fun r hr =>
          closure_minimal
            (fun ⦃y⦄ hy => Classical.byContradiction fun hyK => hy (hs r ⟨hr.left, LE.le.trans_lt hr.right hT1⟩ y hyK))
            (IsCompact.isClosed hK);
        Navier.ConstructionR3.WholeSpaceUniqueness.classical_uniqueness_on_Icc hTpos
          (ContDiffOn.mono h.velocity_smooth hpre) hv (ContDiffOn.mono h.pressure_smooth hpre) hq hK hsupport hev
          (fun r hr => h.divergence_free r ⟨LT.lt.le hr.left, LT.lt.trans hr.right hT1⟩) hdv
          (fun r hr y =>
            Eq.mpr
              (id
                (congr
                  (congrArg Eq
                    (congrFun'
                      (congrArg HAdd.hAdd
                        (congrArg
                          (HSub.hSub
                            (Navier.Construction.ProblemStatement.temporalDerivative u r y +
                              Navier.Construction.ProblemStatement.advection u r y))
                          (one_smul ℝ (Navier.Construction.ProblemStatement.spatialLaplacian u r y))))
                      (Navier.Construction.ProblemStatement.pressureGradient p r y)))
                  (congrFun'
                    (congrArg HAdd.hAdd
                      (congrArg
                        (HSub.hSub
                          (Navier.Construction.ProblemStatement.temporalDerivative v r y +
                            Navier.Construction.ProblemStatement.advection v r y))
                        (one_smul ℝ (Navier.Construction.ProblemStatement.spatialLaplacian v r y))))
                    (Navier.Construction.ProblemStatement.pressureGradient q r y))))
              (Eq.trans (h.navier_stokes r ⟨hr.left, LT.lt.trans hr.right hT1⟩ y) (Eq.symm (hNSv r hr y))))
          fun y => Eq.trans (h.zero_initial_velocity y) (Eq.symm (hvzero y))
  else fun t ht x =>
    have htEq := le_antisymm (LE.le.trans ht.right (le_of_not_gt hTpos)) ht.left;
    Eq.mpr (id (congrArg (fun _a => u (_a, x) = v (_a, x)) htEq))
      (Eq.mpr (id (congrArg (fun _a => _a = v (0, x)) (h.zero_initial_velocity x)))
        (Eq.mpr (id (congrArg (fun _a => 0 = _a) (hvzero x))) (Eq.refl 0)))
